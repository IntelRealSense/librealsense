571  // ---- typed composite-option cast helpers (get_composite_option_as() and friends) -----
572  // Detects whether T has a `.header.version` member via a composed dpp_header - value
573  // structs like rs2_hdrd_control.
574  template< typename U >
575  class has_header_version
576  {
577  template< typename V > static auto test( int ) -> decltype( std::declval< V >().header.version, std::true_type{} );
578  template< typename > static std::false_type test( ... );
579  public:
580  static const bool value = decltype( test< U >( 0 ) )::value;
581  };
582 
583  // Detects whether T is a {min,max,step,def} range wrapper whose ELEMENTS each carry their
584  // own `.header.version` - e.g. rs2_hdrd_control_range. The wrapper itself has no version
585  // field of its own (see rs_hdrd_control.h) - each of its four bounds does.
586  template< typename U >
587  class has_min_header_version
588  {
589  template< typename V > static auto test( int ) -> decltype( std::declval< V >().min.header.version, std::true_type{} );
590  template< typename > static std::false_type test( ... );
591  public:
592  static const bool value = decltype( test< U >( 0 ) )::value;
593  };
594 
595  // T has neither shape - no dpp_header anywhere in it - nothing to check. Every composite-
596  // option struct currently in the SDK has one or the other, but a future one might not
597  // (e.g. a control outside the HKR DPP family entirely).
598  template< typename T >
599  static typename std::enable_if< ! has_header_version< T >::value && ! has_min_header_version< T >::value >::type
600  check_version_supported( const T & )
601  {
602  }
603 
604  // This is what makes the version field worth having: reject anything other than
605  // DPP_HEADER_CURRENT_VERSION up front - including 0, the "never populated" sentinel - rather
606  // than silently misreading a struct laid out by a wire version this build doesn't know.
607  template< typename T >
608  static typename std::enable_if< has_header_version< T >::value >::type
609  check_version_supported( const T & value )
610  {
611  if( value.header.version != DPP_HEADER_CURRENT_VERSION )
612  throw std::runtime_error( "composite option struct has unsupported wire version "
613  + std::to_string( (int)value.header.version ) + " (this SDK build supports version "
614  + std::to_string( (int)DPP_HEADER_CURRENT_VERSION )
615  + ") - wrong struct for this option id, a zero-initialized header about to "
616  "be sent, or a newer device/firmware speaking a wire version this build "
617  "doesn't know yet?" );
618  }
619 
620  // Range wrapper (e.g. rs2_hdrd_control_range) - checks each of the four bounds' own
621  // header.version the same way, since the wrapper carries none of its own.
622  template< typename T >
623  static typename std::enable_if< has_min_header_version< T >::value >::type
624  check_version_supported( const T & value )
625  {
626  check_version_supported( value.min );
627  check_version_supported( value.max );
628  check_version_supported( value.step );
629  check_version_supported( value.def );
630  }

639  std::memcpy( &out, raw.data(), sizeof( T ) );
